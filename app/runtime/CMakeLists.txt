cmake_minimum_required(VERSION 3.19)

set(VCPKG_ROOT "C:/vcpkg" CACHE PATH "Path to vcpkg")
if (NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  if (EXISTS "${VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake")
    set(CMAKE_TOOLCHAIN_FILE "${VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake" CACHE STRING "")
  endif()
endif()

project(open_visual_droplet_sorter LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(CTest)

set(DCAM_SDK_DIR "" CACHE PATH "Path to Hamamatsu DCAM SDK (dcamsdk4)")
set(ONNXRUNTIME_DIR "" CACHE PATH "Path to ONNX Runtime")
set(NIDAQMX_DIR "" CACHE PATH "Path to NI-DAQmx (optional)")

option(ENABLE_NIDAQMX "Enable NI-DAQmx trigger support" ON)
option(BUILD_QT_GUI "Build Qt GUI app" ON)
option(BUILD_QT_GUI_V2 "Build Qt Quick v2 GUI app" OFF)

find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)

if (ONNXRUNTIME_DIR)
  if (NOT ONNXRUNTIME_INCLUDE_DIR)
    find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h PATHS "${ONNXRUNTIME_DIR}/include" NO_DEFAULT_PATH)
  endif()
  if (NOT ONNXRUNTIME_LIB)
    find_library(ONNXRUNTIME_LIB onnxruntime PATHS "${ONNXRUNTIME_DIR}/lib" "${ONNXRUNTIME_DIR}/lib64" NO_DEFAULT_PATH)
  endif()
else()
  if (NOT ONNXRUNTIME_INCLUDE_DIR)
    find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h)
  endif()
  if (NOT ONNXRUNTIME_LIB)
    find_library(ONNXRUNTIME_LIB onnxruntime)
  endif()
endif()
if (NOT ONNXRUNTIME_INCLUDE_DIR OR NOT ONNXRUNTIME_LIB)
  message(FATAL_ERROR "ONNX Runtime not found. Set ONNXRUNTIME_DIR to a valid install path.")
endif()

set(DCAM_INCLUDE_DIR "${DCAM_SDK_DIR}/inc")
set(DCAM_LIB_DIR "${DCAM_SDK_DIR}/lib/win64")

if (BUILD_QT_GUI)
  add_subdirectory(desktop_app)
endif()

if (BUILD_QT_GUI_V2)
  set(_OPENDSS_PARENT_PROJECT_NAME "${CMAKE_PROJECT_NAME}")
  get_property(_OPENDSS_QML_IMPORT_PATH_EXISTED CACHE QML_IMPORT_PATH PROPERTY VALUE SET)
  if (_OPENDSS_QML_IMPORT_PATH_EXISTED)
    get_property(_OPENDSS_QML_IMPORT_PATH_VALUE CACHE QML_IMPORT_PATH PROPERTY VALUE)
    get_property(_OPENDSS_QML_IMPORT_PATH_TYPE CACHE QML_IMPORT_PATH PROPERTY TYPE)
    get_property(_OPENDSS_QML_IMPORT_PATH_HELP CACHE QML_IMPORT_PATH PROPERTY HELPSTRING)
  endif()

  set(CMAKE_PROJECT_NAME Desktop_app_v2App)
  add_subdirectory(Desktop_app_v2)
  set(CMAKE_PROJECT_NAME "${_OPENDSS_PARENT_PROJECT_NAME}")

  if (_OPENDSS_QML_IMPORT_PATH_EXISTED)
    set_property(CACHE QML_IMPORT_PATH PROPERTY VALUE "${_OPENDSS_QML_IMPORT_PATH_VALUE}")
    set_property(CACHE QML_IMPORT_PATH PROPERTY TYPE "${_OPENDSS_QML_IMPORT_PATH_TYPE}")
    set_property(CACHE QML_IMPORT_PATH PROPERTY HELPSTRING "${_OPENDSS_QML_IMPORT_PATH_HELP}")
  else()
    unset(QML_IMPORT_PATH CACHE)
  endif()

  unset(_OPENDSS_PARENT_PROJECT_NAME)
  unset(_OPENDSS_QML_IMPORT_PATH_EXISTED)
  unset(_OPENDSS_QML_IMPORT_PATH_VALUE)
  unset(_OPENDSS_QML_IMPORT_PATH_TYPE)
  unset(_OPENDSS_QML_IMPORT_PATH_HELP)
endif()
